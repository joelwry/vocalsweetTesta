cmake_minimum_required(VERSION 3.22)

project(VocalSweet VERSION 1.0.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(VOCALSWEET_BUILD_TESTS "Build the DSP / plugin test executables" ON)
option(VOCALSWEET_STANDALONE "Also build the Standalone application" ON)
set(VOCALSWEET_JUCE_DIR "" CACHE PATH "Path to a JUCE 8 checkout (empty = download JUCE 8.0.4)")

# ---------------------------------------------------------------------------------------------
# JUCE
if(VOCALSWEET_JUCE_DIR)
    add_subdirectory(${VOCALSWEET_JUCE_DIR} JUCE)
else()
    include(FetchContent)
    FetchContent_Declare(JUCE
        GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
        GIT_TAG 8.0.4
        GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(JUCE)
endif()

# ---------------------------------------------------------------------------------------------
# Plugin
set(VS_FORMATS VST3)
if(VOCALSWEET_STANDALONE)
    list(APPEND VS_FORMATS Standalone)
endif()

# The VST3 manifest (moduleinfo.json) is generated by *running* a helper that loads the plugin.
# When cross-compiling that only works if an emulator (e.g. wine) is configured.
set(VS_AUTO_MANIFEST TRUE)
if(CMAKE_CROSSCOMPILING AND NOT CMAKE_CROSSCOMPILING_EMULATOR)
    set(VS_AUTO_MANIFEST FALSE)
endif()

juce_add_plugin(VocalSweet
    PRODUCT_NAME "VocalSweet"
    COMPANY_NAME "VocalSweet Audio"
    COMPANY_WEBSITE "https://example.invalid/vocalsweet"
    BUNDLE_ID com.vocalsweet.vocalsweet
    PLUGIN_MANUFACTURER_CODE Vswt
    PLUGIN_CODE Vsw1
    FORMATS ${VS_FORMATS}
    IS_SYNTH FALSE
    NEEDS_MIDI_INPUT FALSE
    NEEDS_MIDI_OUTPUT FALSE
    IS_MIDI_EFFECT FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS FALSE
    VST3_CATEGORIES Fx Dynamics EQ
    VST3_AUTO_MANIFEST ${VS_AUTO_MANIFEST}
    COPY_PLUGIN_AFTER_BUILD FALSE)

juce_generate_juce_header(VocalSweet)

target_sources(VocalSweet PRIVATE
    Source/PluginProcessor.cpp
    Source/PluginEditor.cpp
    Source/PresetManager.cpp
    Source/ui/LookAndFeel.cpp
    Source/ui/Widgets.cpp)

juce_add_binary_data(VocalSweetBinaryData
    HEADER_NAME BinaryData.h
    NAMESPACE BinaryData
    SOURCES
        Resources/fonts/Inter-Regular.ttf
        Resources/fonts/Inter-SemiBold.ttf
        Resources/fonts/Inter-Bold.ttf)

target_compile_definitions(VocalSweet PUBLIC
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0
    JUCE_DISPLAY_SPLASH_SCREEN=0
    JUCE_REPORT_APP_USAGE=0
    JUCE_MODAL_LOOPS_PERMITTED=0)

target_include_directories(VocalSweet PUBLIC Source)

target_link_libraries(VocalSweet
    PRIVATE
        VocalSweetBinaryData
        juce::juce_audio_utils
        juce::juce_dsp
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_lto_flags
        juce::juce_recommended_warning_flags)

# ---------------------------------------------------------------------------------------------
# Tests and tools (host builds only)
if(VOCALSWEET_BUILD_TESTS AND NOT CMAKE_CROSSCOMPILING)
    enable_testing()

    # Pure DSP tests - no JUCE dependency
    add_executable(VocalSweetDspTests tests/DspTests.cpp)
    target_include_directories(VocalSweetDspTests PRIVATE Source tests)
    target_compile_options(VocalSweetDspTests PRIVATE -O2 -Wall -Wextra)
    add_test(NAME dsp COMMAND VocalSweetDspTests ${CMAKE_BINARY_DIR}/test-output)

    # Offline renderer: synthetic vocal test signal through every preset -> WAV files
    add_executable(VocalSweetRender tests/RenderPresets.cpp)
    target_include_directories(VocalSweetRender PRIVATE Source tests)
    target_compile_options(VocalSweetRender PRIVATE -O2)

    # Plugin-level tests (processor, parameters, state, presets, editor snapshots).
    # The plugin sources are compiled directly into this console app (a plugin-independent host).
    juce_add_console_app(VocalSweetPluginTests PRODUCT_NAME "VocalSweetPluginTests")
    juce_generate_juce_header(VocalSweetPluginTests)
    target_sources(VocalSweetPluginTests PRIVATE
        tests/PluginTests.cpp
        Source/PluginProcessor.cpp
        Source/PluginEditor.cpp
        Source/PresetManager.cpp
        Source/ui/LookAndFeel.cpp
        Source/ui/Widgets.cpp)
    target_include_directories(VocalSweetPluginTests PRIVATE Source)
    target_compile_definitions(VocalSweetPluginTests PRIVATE
        JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=0
        JucePlugin_Name="VocalSweet" JucePlugin_VersionString="${PROJECT_VERSION}")
    target_link_libraries(VocalSweetPluginTests PRIVATE
        VocalSweetBinaryData
        juce::juce_audio_utils
        juce::juce_dsp
        juce::juce_recommended_config_flags)
    add_test(NAME plugin COMMAND VocalSweetPluginTests ${CMAKE_BINARY_DIR}/test-output)
endif()
